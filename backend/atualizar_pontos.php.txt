<?php
header("Content-Type: application/json; charset=UTF-8");
include_once 'conexao.php';

$dados = json_decode(file_get_contents("php://input"));

if(!empty($dados->nome) && isset($dados->pontos)) {
    // Atualiza somando os pontos novos aos pontos que já existem na tabela
    $query = "UPDATE jogadores SET pontos = pontos + :pontos_novos WHERE nome = :nome";
    $stmt = $conn->prepare($query);
    
    $stmt->bindParam(":pontos_novos", $dados->pontos);
    $stmt->bindParam(":nome", $dados->nome);
    
    if($stmt->execute()) {
        http_response_code(200); // OK
        echo json_encode(["mensagem" => "Pontuação atualizada."]);
    } else {
        http_response_code(503); 
        echo json_encode(["mensagem" => "Não foi possível atualizar os pontos."]);
    }
} else {
    http_response_code(400);
    echo json_encode(["mensagem" => "Dados incompletos."]);
}
?>
